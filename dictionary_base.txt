# Встроенный словарь замен VoiceToText: марки разделов проектной и рабочей документации.
#
# Этот файл — часть программы, он сохраняется в git и не потеряется.
# Ваши личные замены — в файле dictionary.txt (папка %APPDATA%\VoiceToText).
# Личный словарь применяется ПЕРВЫМ, поэтому им можно переопределить любую строку отсюда.
#
# Формат: как расслышала модель = как надо написать
# Пробелы и дефисы внутри левой части не важны: строка "а эр = АР"
# сработает и на "аэр", и на "а эр", и на "а-эр".
# Строки, начинающиеся с #, не работают — это комментарии.
# Чтобы отключить замену, поставьте # в начале строки. Чтобы включить — уберите #.

# --- Составные марки идут первыми, иначе их «съедят» короткие ---
ка эм дэ = КМД
эн вэ ка = НВК
э о эм = ЭОМ
пэ зэ у = ПЗУ
пзу = ПЗУ
пэ о эс = ПОС
эс эс = СС

# --- Архитектура и конструкции ---
а эр = АР
air = АР
а эс = АС
ка эр = КР
кэр = КР
ка жэ = КЖ
ка же = КЖ
ка эм = КМ

# --- Инженерные разделы ---
э эс = ЭС
эом = ЭОМ
вэ ка = ВК
о вэ = ОВ
о ве = ОВ
тэ ха = ТХ
те ха = ТХ
пос = ПОС

# --- Спорные строки: включайте, только если мешает ---
# "века" — обычное слово (века истории), поэтому по умолчанию выключено:
# века = ВК
# "под" — предлог, автозамену делать нельзя; марку ПОД пишите руками.
# "см" — это ещё и сантиметры, поэтому марки СМ здесь нет.
